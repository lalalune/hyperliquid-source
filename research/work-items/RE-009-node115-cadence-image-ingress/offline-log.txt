from pathlib import Path

from stage import DECODED_SIZE, read_checkpoint, stage
from test_stage import fixture

def digest(value):
    return sha256(value).hexdigest()

def state(height, root_field):
    with TemporaryDirectory() as directory:
        raw = fixture(height, root_field)
        source = Path(directory) / "state.rmp"
        source.write_bytes(raw)
        cadence = read_checkpoint(source, digest(raw), height)
    decoded = bytearray(DECODED_SIZE)
    pack_into("<Q", decoded, 0x28, height)
    staged = stage(decoded, digest(decoded), height, cadence)
    evm = bytes(EVM_BLOCK_SIZE)
    final = assemble(staged, evm, digest(staged), digest(evm), height, bytes(4), bytes(3))
    compact_cadence, = unpack_from("<Q", final, 0x39c8)
    assert compact_cadence == cadence
    print(height, "materialized cadence", compact_cadence, "image SHA-256", digest(final))
    return SelectorState(0, 0, 0, compact_cadence)
