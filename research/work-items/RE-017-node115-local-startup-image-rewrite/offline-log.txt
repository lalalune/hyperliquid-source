from hashlib import sha256
from os import environ
from pathlib import Path
from struct import unpack_from
from rewrite import rewrite_local

fixtures = Path(environ["FIXTURES"])
out = Path(environ["OUT"])
for case, expected_round in [("absent", 99), ("future", 99), ("equal", 105), ("past", None)]:
    before = (fixtures / (case + ".compact")).read_bytes()
    try:
        result = rewrite_local(before, sha256(before).hexdigest(), 41)
    except ValueError as error:
        assert case == "past" and "scheduled freeze 40" in str(error)
    else:
        assert expected_round is not None
        assert result.original_pair == ((1 << 64) - 1, 99)
        assert result.replay_pair == ((1 << 64) - 1, expected_round)
        assert unpack_from("<QQ", result.image, 0x18) == result.replay_pair
        assert result.image[0x28:] == before[0x28:]
        (out / (case + ".output.bin")).write_bytes(result.image)
    print("PASS:", case, "after upstream named RMP and compact assembly")
