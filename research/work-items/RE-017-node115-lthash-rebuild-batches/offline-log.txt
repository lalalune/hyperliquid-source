use node115_abci::lt_hash::{LtHash, LT_HASH_RECORD_BYTES};
use std::io::Write;

fn batch(lane: u16, elements: u64, bytes: u64) -> [u8; LT_HASH_RECORD_BYTES] {
    let mut record = [0; LT_HASH_RECORD_BYTES];
    for chunk in record[..0x800].chunks_exact_mut(2) {
        chunk.copy_from_slice(&lane.to_be_bytes());
    }
    record[0x800..0x808].copy_from_slice(&elements.to_le_bytes());
    record[0x808..0x810].copy_from_slice(&bytes.to_le_bytes());
    record
}

fn main() {
    let accounts = LtHash::from_rebuild_batches([
        &batch(0xffff, u64::MAX, 15), &batch(2, 2, 9),
    ]);
    let contracts = LtHash::from_rebuild_batches([&batch(0x104, 3, u64::MAX)]);
    let storage = LtHash::from_rebuild_batches(std::iter::empty());
    for row in [accounts, contracts, storage] {
        std::io::stdout().lock().write_all(&row.to_record_bytes()).unwrap();
    }
}
