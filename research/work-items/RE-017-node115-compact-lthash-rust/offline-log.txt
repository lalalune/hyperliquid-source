    for (index, word) in hashes.accounts_hash.hash.iter().copied().enumerate() {
        assert_eq!(word, (index % 256) as u8);

    assert_eq!(hashes.storage_hash.hash, [51; 2048]);
    assert_eq!((hashes.storage_hash.n_elements, hashes.storage_hash.n_bytes), (0, u64::MAX));
    let (accounts, contracts, storage) = hashes.checksums();
    assert_ne!(accounts, contracts);
    assert_ne!(accounts, storage);
    assert_ne!(contracts, storage);
    println!("PASS: constructed compact fixture decoded into all three Rust LtHash fields");
}
