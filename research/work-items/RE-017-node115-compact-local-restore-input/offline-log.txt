use node115_abci_checkpoint_replay_fixture::{
    CompactLocalRestoreInput, HardforkWords, LocalRestoreError, restore_local_checkpoint,
};
use std::path::PathBuf;
fn main() {
    let base = PathBuf::from(std::env::args_os().nth(1).unwrap());
    for (case, expected) in [("absent", Some(99)), ("future", Some(99)), ("equal", Some(105)), ("past", None)] {
        let bytes = std::fs::read(base.join(format!("{case}.compact"))).unwrap();
        let restored = CompactLocalRestoreInput::from_checkpoint(&bytes, 41).unwrap();
        let original = HardforkWords {
            version: restored.hardfork_version,
            round: restored.hardfork_round,
        };
        let outcome = restore_local_checkpoint(
            &base.join("checkpoint"), 41, restored.height, restored.context_round,
            original, restored.scheduled_freeze,
        );
        if let Some(round) = expected {
            let result = outcome.unwrap();
            assert_eq!(result.startup_pair, original);
            assert_eq!(result.state.loader_seq, u64::MAX);
            assert_eq!(result.state.copied_round, round);
        } else {
            assert_eq!(outcome, Err(LocalRestoreError::PastFreeze { target: 40, height: 41 }));
        }
        println!("PASS: compact checkpoint -> local startup input -> {case} fixture outcome");
    }
}
